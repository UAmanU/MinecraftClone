cmake_minimum_required(VERSION 3.20)
project(MinecraftCodexClone LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RAYLIB_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../Libraries/raylib_files")
add_library(raylib STATIC IMPORTED)
set_target_properties(raylib PROPERTIES IMPORTED_LOCATION "${RAYLIB_DIR}/libraylib.a" INTERFACE_INCLUDE_DIRECTORIES "${RAYLIB_DIR}")
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_executable(MinecraftCodexClone ${SOURCES})
target_include_directories(MinecraftCodexClone PRIVATE include)
target_link_libraries(MinecraftCodexClone PRIVATE raylib winmm gdi32 opengl32)
